// Raid Welcome for Streamer.bot, free from itsCRA1G (itscra1g.github.io/raid-welcome)
// Trigger: Twitch > Channel > Raid. Sub-action: Core > C# > Execute C# Code.
using System;

public class CPHInline
{
    // ===== Change these to suit your stream =====

    // Welcome messages. One is picked at random each raid, never the same twice in a row.
    // {user} = the raider's name, {viewers} = how many they brought.
    private static readonly string[] Welcomes = new[]
    {
        "Welcome in, raiders! Huge thanks to {user} for bringing {viewers} people over. Grab a seat and say hi!",
        "RAID! {user} just landed with {viewers} viewers. Make yourselves at home, everyone!",
        "Look who it is! {user} and {viewers} friends have arrived. Thanks so much for the raid!"
    };

    // Shoutout lines, posted a few seconds after the welcome. {link} = the raider's channel.
    private static readonly string[] Shoutouts = new[]
    {
        "Go show {user} some love, they were live just now: {link}",
        "{user} is a top streamer, give them a follow: {link}",
        "Check out {user} over at {link}"
    };

    // Raids smaller than this are ignored. 1 = welcome every raid.
    private const int MinViewers = 1;

    // Seconds to wait between the welcome and the shoutout line.
    private const int ShoutoutDelaySeconds = 3;

    // Also send Twitch's own /shoutout? Your bot account must be a mod for this.
    private const bool NativeShoutout = true;

    // ===== Nothing to change below here =====

    public bool Execute()
    {
        string user = Arg("user", Arg("userName", "someone"));
        string login = Arg("userName", user).ToLower();
        int viewers = 0;
        int.TryParse(Arg("viewers", "0"), out viewers);

        if (viewers < MinViewers)
            return true;

        string welcome = Pick(Welcomes, "raidwelcome_lastWelcome")
            .Replace("{user}", user)
            .Replace("{viewers}", viewers.ToString());
        CPH.SendMessage(welcome);

        CPH.Wait(ShoutoutDelaySeconds * 1000);

        string shout = Pick(Shoutouts, "raidwelcome_lastShout")
            .Replace("{user}", user)
            .Replace("{link}", "https://twitch.tv/" + login);
        CPH.SendMessage(shout);

        if (NativeShoutout)
        {
            try { CPH.TwitchSendShoutoutByLogin(login); }
            catch (Exception e) { CPH.LogInfo("Raid Welcome: Twitch shoutout skipped (" + e.Message + ")"); }
        }
        return true;
    }

    private string Arg(string key, string fallback)
    {
        object v;
        if (args.TryGetValue(key, out v) && v != null && v.ToString().Length > 0)
            return v.ToString();
        return fallback;
    }

    private string Pick(string[] list, string varName)
    {
        if (list.Length == 1) return list[0];
        int last = CPH.GetGlobalVar<int>(varName, false);
        int i;
        do { i = new Random().Next(list.Length); } while (i == last - 1);
        CPH.SetGlobalVar(varName, i + 1, false);
        return list[i];
    }
}

Set it up in Streamer.bot

  1. Open the Actions tab, right-click in the Actions list and pick Add. Call it Raid Welcome and press OK.
  2. In the Triggers box, right-click and pick Twitch → Channel → Raid. Leave its settings as they are and press OK.
  3. In the Sub-Actions box, right-click and pick Core → C# → Execute C# Code.
  4. Delete everything in the code window, paste the code from step 1, then press Compile. It should say Compiled successfully.
  5. Press Save and Compile, and make sure the action is ticked as enabled.
Test it without waiting for a raid: right-click the Raid trigger and pick Test Trigger. Your bot posts the messages in your chat.
For the Twitch shoutout, the account that sends your chat messages must be a moderator in your channel. If you use a separate bot account, type /mod yourbotname in your chat.

Make it yours

Everything you'd want to change is at the top of the code, above Nothing to change below here. Edit it in the code window and press Save and Compile again.

SettingWhat it does
WelcomesThe welcome messages. Add as many as you like, each in quotes with a comma after it. {user} becomes the raider's name and {viewers} how many they brought.
ShoutoutsThe shoutout lines. {link} becomes the raider's Twitch link.
MinViewersRaids smaller than this are ignored. Set it to 5, say, to skip one-person raids.
ShoutoutDelaySecondsThe gap between the welcome and the shoutout line.
NativeShoutoutSet to false to turn off Twitch's own shoutout.

Keep chat messages under 500 characters, Twitch's limit.

If something's not right

What you seeWhat to do
It won't compileMake sure you pasted over everything in the code window, and that any message you edited still has quotes at both ends and a comma after it.
Messages post, but no Twitch shoutoutYour bot account isn't a mod, or Twitch's limit kicked in: one shoutout every 2 minutes, and the same streamer once an hour.
Nothing happens on a raidCheck the action is enabled, the trigger is Twitch → Channel → Raid, and the raid is at least MinViewers.
